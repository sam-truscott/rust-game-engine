# Single click to look at things, double click to interact with them.
# Run: target/debug/game.exe --auto games/pub/play/hotspots.txt --out screenshots
# Positions are screen points (320x180, y up): the room is above the 26-high panel, so a room point
# (x, y) is clicked at (x, y + 26). Two `click`s in a row are a double click (the time step is fixed).
wait 3
move 240 96                 # over the bar
click 240 96                # single click: look at the bar
wait 0.6
shot h1_walking_to_bar
wait 4
shot h2_looking             # standing at the bar, speaking
where                       # (240, 40), speaking, handler still running
wait 4
where                       # (240, 40), nothing running

click 96 146                # look at the window
wait 0.3
click 96 146                # second click: cuts the look short and interacts instead
wait 4
shot h3_window_interact     # at the window's spot (96, 80), "It doesn't open."
wait 3
where                       # (96, 80), nothing running

click 240 96
click 240 96                # the bar again
wait 4
shot h4_bar_interact        # "The till is locked."
wait 3
click 160 20                # on the panel: nothing happens
where                       # player has not moved: (240, 40)
quit
