# Hover over things, pick verbs and use them on the counter and the window.
# Run: target/debug/game.exe --auto play/hotspots.txt --out screenshots
# Positions are screen points (320x180, y up): the room is above the 40-high panel, so a room point
# (x, y) is clicked at (x, y + 40).
# Verb buttons (centres): Walk to 28,32  Look at 80,32  Pick up 132,32  Use 184,32  Open 236,32  Talk to 288,32
wait 3
move 160 110                # over the counter
shot h1_hover               # Walk to highlighted
click 80 32                 # Look at
shot h2_look_selected       # Look at highlighted
click 160 110               # the counter
wait 0.6
shot h3_walking_to_counter
wait 2.5
shot h4_looking             # standing in front of the counter, speaking
where                       # (160, 30), speaking, handler still running
wait 4
where                       # verb back to WalkTo, nothing running

click 236 32                # Open
click 256 115               # the window
wait 2.5
shot h5_window_open         # at the window's spot (256, 40), "It's only a frame."
wait 3
where                       # (256, 40), nothing running

click 288 32                # Talk to: the counter has no handler for it
click 160 110
wait 2.5
shot h6_default_reply       # "I can't do that."
wait 3
click 106 32                # on the panel, in the gap between two buttons: nothing happens
where                       # verb still WalkTo, player has not moved
quit
